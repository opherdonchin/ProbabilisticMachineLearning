RANDOM_SEED = 2027

CELL 5
n1 = 3
n2 = 4

mu1 = np.array([-0.5, 0.5, 1.0])
mu2 = np.array([0.0, 0.5, -0.5, 1.0])
mu = np.concatenate([mu1, mu2])

sigma1 = 1.0
sigma2 = 1.2
rho1 = 0.70
rho2 = 0.80
kappa = 0.90

Sigma11 = sigma1**2 * (
    (1 - rho1) * np.eye(n1)
    + rho1 * np.ones((n1, n1))
)

Sigma22 = sigma2**2 * (
    (1 - rho2) * np.eye(n2)
    + rho2 * np.ones((n2, n2))
)

Sigma12 = (
    kappa
    * sigma1
    * sigma2
    * np.sqrt(rho1 * rho2)
    * np.ones((n1, n2))
)

Sigma21 = Sigma12.T


CELL 12
Sigma = np.block([
    [Sigma11, Sigma12],
    [Sigma21, Sigma22],
])

print(Sigma.shape)


CELL 17
sd = np.sqrt(np.diag(Sigma))
Corr = Sigma / np.outer(sd, sd)

plt.figure(figsize=(6, 5))
plt.imshow(Corr, vmin=-1, vmax=1)
plt.colorbar(label="correlation")
plt.axvline(n1 - 0.5)
plt.axhline(n1 - 0.5)
plt.xticks(range(n1 + n2))
plt.yticks(range(n1 + n2))
plt.title("Correlation structure of the joint Gaussian")
plt.tight_layout()
plt.show()

CELL 43
x1_posterior = azb.extract(
    idata,
    var_names="x_unobserved",
)

sample_dim = "sample"
component_dim = next(
    dim for dim in x1_posterior.dims
    if dim != sample_dim
)

posterior_samples = (
    x1_posterior
    .transpose(sample_dim, component_dim)
    .values
)

positions = np.arange(n1) + 1

def plot_component_posteriors(samples, truth, title):
    plt.figure(figsize=(8, 4.5))
    plt.violinplot(
        [samples[:, i] for i in range(samples.shape[1])],
        positions=positions,
        showmeans=True,
        showextrema=False,
    )
    plt.scatter(
        positions,
        truth,
        marker="D",
        s=45,
        label="Hidden true value",
    )
    plt.xticks(
        positions,
        [f"x1[{i}]" for i in range(n1)],
    )
    plt.ylabel("value")
    plt.title(title)
    plt.legend()
    plt.tight_layout()
    plt.show()

plot_component_posteriors(
    posterior_samples,
    x1_true,
    "PyMC posterior",
)

mu_z = mu1

CELL 51
W = np.linalg.solve(
    Sigma11,
    Sigma12,
).T

W


CELL 53
b = mu2 - W @ mu1
b


CELL 57
Omega = (
    Sigma22
    - Sigma21
    @ np.linalg.solve(
        Sigma11,
        Sigma12,
    )
)

Omega


CELL 59
m = W @ mu_z + b

S = (
    W
    @ Sigma_z
    @ W.T
    + Omega
)

print("Mean check:", np.allclose(m, mu2))
print("Covariance check:", np.allclose(S, Sigma22))


CELL 65
lgs_mean = (
    mu_z
    + Sigma_z
    @ W.T
    @ np.linalg.solve(
        S,
        y - m,
    )
)

lgs_mean


CELL 67
lgs_cov = (
    Sigma_z
    - Sigma_z
    @ W.T
    @ np.linalg.solve(
        S,

)

lgs_cov
